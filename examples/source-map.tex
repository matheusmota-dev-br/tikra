\documentclass{standalone}
\usepackage{tikz,pgfplots}
\newcommand{\card}[2]{\node[draw, fill=blue!10] at (#1,0) {#2};}
\begin{document}
\begin{tikzpicture}
  \draw[help lines] (0,-1) grid (6,1);
  \card{1}{macro}
  \card{4}{call}

    \fill[red] (\x,-1) circle (2pt);
  }
  \draw[->, thick, blue]
    (0,1) .. controls (2,2) and (4,0) ..
    (6,1);
\end{tikzpicture}
