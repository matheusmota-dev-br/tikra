      \pgfmathsetmacro\cx{1.5*\r*\i}

        (\cx,\cy) +(0:\r) -- +(60:\r) -- +(120:\r) -- +(180:\r) -- +(240:\r) -- +(300:\r) -- cycle;
    }
  }
