% Made up for the tikra benchmark.
% Euler's formula on the unit circle: e^{i theta} = cos theta + i sin theta, and at theta = pi
% the identity e^{i pi} + 1 = 0.
\begin{tikzpicture}[scale=2.6, >=Stealth]
  \draw[gray!30, very thin] (-1.3,-1.3) grid[step=0.25] (1.3,1.3);
  \draw[->, thick] (-1.4,0) -- (1.5,0) node[right] {$\mathrm{Re}$};
  \draw[->, thick] (0,-1.3) -- (0,1.45) node[above] {$\mathrm{Im}$};
  \draw[blue!60!black, very thick] (0,0) circle (1);
  \coordinate (P) at (50:1);
  \fill[orange!25] (0,0) -- (0.3,0) arc (0:50:0.3) -- cycle;
  \node[orange!70!black] at (25:0.42) {$\theta$};
  \draw[->, very thick, red!70!black] (0,0) -- (P) node[above right] {$e^{i\theta}$};
  \draw[dashed, red!60!black] (P) -- (P |- 0,0) node[below] {$\cos\theta$};
  \draw[dashed, red!60!black] (P) -- (P -| 0,0) node[left] {$i\sin\theta$};
  \fill[red!70!black] (P) circle (0.9pt);
  \fill[purple] (-1,0) circle (1.2pt) node[below left] {$e^{i\pi} = -1$};
  \draw[->, purple, thick] (0.6,0) arc (0:180:0.6);
  \node[draw, rounded corners, fill=yellow!20, font=\large, anchor=north west] at (0.35,-0.55) {$e^{i\pi} + 1 = 0$};
\end{tikzpicture}
