\begin{tikzpicture}[x={(0.9cm,-0.35cm)}, y={(0.75cm,0.4cm)}, z={(0cm,0.9cm)}]
  \newcommand\block[4]{% x, y, height, color
    \fill[#4!80!black] (#1,#2,0) -- (#1+0.7,#2,0) -- (#1+0.7,#2,#3) -- (#1,#2,#3) -- cycle;
    \fill[#4!55!black] (#1+0.7,#2,0) -- (#1+0.7,#2+0.7,0) -- (#1+0.7,#2+0.7,#3) -- (#1+0.7,#2,#3) -- cycle;
    \fill[#4!45] (#1,#2,#3) -- (#1+0.7,#2,#3) -- (#1+0.7,#2+0.7,#3) -- (#1,#2+0.7,#3) -- cycle;
    \draw[black!60, line join=round] (#1,#2,0) -- (#1+0.7,#2,0) -- (#1+0.7,#2+0.7,0)
      -- (#1+0.7,#2+0.7,#3) -- (#1,#2+0.7,#3) -- (#1,#2,#3) -- cycle
      (#1,#2,#3) -- (#1+0.7,#2,#3) -- (#1+0.7,#2+0.7,#3) (#1+0.7,#2,#3) -- (#1+0.7,#2,0);
  }
  \draw[gray!50] (0,3,0) -- (0,0,0) -- (4,0,0);
  \draw[gray!50] (0,3,0) -- (4,3,0) -- (4,0,0);
  \draw[gray!50] (0,3,0) -- (0,3,4);
  \foreach \zz in {1,2,3} { \draw[gray!30] (0,3,\zz) -- (4,3,\zz); \node[left, font=\tiny] at (0,3,\zz) {\zz}; }
  \foreach \y/\c in {2/teal, 1/orange, 0/blue} {
    \foreach \x/\h in {0/1.4, 1/2.6, 2/2, 3/1} {
      \pgfmathsetmacro\hh{\h + 0.4*\y}
      \block{\x+0.15}{\y+0.15}{\hh}{\c}
    }
  }
  \foreach \x/\q in {0/Q1, 1/Q2, 2/Q3, 3/Q4} { \node[anchor=north, font=\small] at (\x+0.5,0,0) {\q}; }
  \foreach \y/\r in {0/North, 1/South, 2/West} { \node[anchor=west, font=\small] at (4.1,\y+0.5,0) {\r}; }
\end{tikzpicture}
