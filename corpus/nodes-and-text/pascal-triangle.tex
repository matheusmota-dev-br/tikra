% Made up for the tikra benchmark.
% Pascal's triangle: each number the sum of the two above it; the odd ones colored, which draws
% a Sierpinski triangle.
\begin{tikzpicture}[x=0.75cm, y=0.65cm, cell/.style={circle, minimum size=6.6mm, inner sep=0pt, font=\footnotesize\bfseries}]
  \foreach \n in {0,...,11} {
    \foreach \k in {0,...,\n} {
      \pgfmathtruncatemacro\c{factorial(\n)/(factorial(\k)*factorial(\n-\k))}
      \pgfmathtruncatemacro\odd{mod(\c,2)}
      \ifnum\odd=1
        \node[cell, fill=orange!70, draw=orange!60!black] at (\k-\n/2, -\n) {\c};
      \else
        \node[cell, fill=blue!12, draw=blue!30, text=blue!50!black] at (\k-\n/2, -\n) {\c};
      \fi
    }
  }
  \node[font=\Large\bfseries] at (0,1.3) {Pascal's triangle};
  \node[font=\small] at (0,-12.4) {$\binom{n}{k} = \binom{n-1}{k-1} + \binom{n-1}{k}$};
\end{tikzpicture}
